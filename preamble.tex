% Compact chapter headings suit a course report and leave room for examples.
% Keep report-class numbering (including appendices), with no extra package.
\makeatletter
\renewcommand{\@makechapterhead}[1]{%
  \vspace*{8pt}%
  {\parindent\z@\raggedright\normalfont
    \ifnum\c@secnumdepth>\m@ne
      {\large\color{reportblue}\@chapapp\space\thechapter\par}%
      \vskip8pt
    \fi
    \LARGE\bfseries #1\par\nobreak\vskip24pt
  }}
\renewcommand{\@makeschapterhead}[1]{%
  \vspace*{8pt}%
  {\parindent\z@\raggedright\normalfont
    \LARGE\bfseries #1\par\nobreak\vskip24pt
  }}
\makeatother
